cmake_minimum_required(VERSION 3.22)
project(totk_package_import C)
set(H ${CMAKE_CURRENT_LIST_DIR}/../../third_party/hactool)
file(GLOB CRYPTO ${H}/mbedtls/library/*.c)
add_library(package_crypto STATIC ${CRYPTO})
target_include_directories(package_crypto PUBLIC ${H}/mbedtls/include)
set_target_properties(package_crypto PROPERTIES POSITION_INDEPENDENT_CODE ON)
file(GLOB HAC ${H}/*.c)
add_executable(package_import ${HAC} slice.c)
target_include_directories(package_import PRIVATE ${CMAKE_CURRENT_LIST_DIR} ${H}/mbedtls/include)
target_compile_definitions(package_import PRIVATE _FILE_OFFSET_BITS=64 _DEFAULT_SOURCE)
target_link_libraries(package_import PRIVATE package_crypto)
# Android only executes packaged native files, so this PIE has a library filename.
set_target_properties(package_import PROPERTIES OUTPUT_NAME libpackage_import.so)
if(ANDROID)
    target_link_options(package_import PRIVATE -Wl,-z,max-page-size=16384)
endif()
